# Автоматизация: Google Forms → Битрикс24 → welcome-письмо import gspread import requests BITRIX = "https://lidereco.bitrix24.ru/rest/1/TOKEN/" def process_applications(): sheet = gc.open("Регистрация").worksheet("Ответы") for row in sheet.get_all_records(): # Создаём сделку в Битрикс24 deal = requests.post( f"{BITRIX}crm.deal.add", json={"fields": { "TITLE": f"Заявка: {row['ФИО']}", "CONTACT_NAME": row["ФИО"], "PHONE": [{"VALUE": row["Телефон"]}], "EMAIL": [{"VALUE": row["Email"]}] }} ).json().get("result") # Welcome-письмо requests.post(f"{BITRIX}crm.activity.add", json={ "fields": { "OWNER_TYPE_ID": "DEAL", "OWNER_ID": deal, "TYPE_ID": "EMAIL", "SUBJECT": "Добро пожаловать!" } }) print(f"✅ {row['ФИО']} → сделка #{deal}") process_applications()
