import requests
import xml.etree.ElementTree as ET
import json
import csv
from datetime import datetime

# API «Фабриканта»
FABRIKANT_API = "https://api.fabrikant.ru/v2/"
API_KEY = "YOUR_API_KEY"

def fetch_tender_data(tender_id: str) -> dict:
    """
    Выгрузка данных по завершённому тендеру
    """
    response = requests.get(
        f"{FABRIKANT_API}protocols/{tender_id}",
        headers={"Authorization": f"Bearer {API_KEY}"},
        params={"format": "xml"}
    )
    
    # Парсим XML
    root = ET.fromstring(response.content)
    
    tender_data = {
        "tender_id": tender_id,
        "title": root.find(".//Name").text,
        "winner": {
            "name": root.find(".//Winner/Name").text,
            "inn": root.find(".//Winner/INN").text,
            "kpp": root.find(".//Winner/KPP").text
        },
        "contract": {
            "amount": float(root.find(".//Contract/Amount").text),
            "subject": root.find(".//Contract/Subject").text,
            "date_start": root.find(".//Contract/DateStart").text,
            "date_end": root.find(".//Contract/DateEnd").text
        },
        "protocol_date": root.find(".//ProtocolDate").text
    }
    
    return tender_data

def export_to_csv(tenders: list, filename: str):
    """
    Экспорт в CSV для импорта в 1С
    """
    fieldnames = [
        "tender_id", "title", "winner_name", "winner_inn", 
        "winner_kpp", "contract_amount", "contract_subject",
        "date_start", "date_end"
    ]
    
    with open(filename, 'w', newline='', encoding='utf-8-sig') as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        
        for t in tenders:
            writer.writerow({
                "tender_id": t["tender_id"],
                "title": t["title"],
                "winner_name": t["winner"]["name"],
                "winner_inn": t["winner"]["inn"],
                "winner_kpp": t["winner"]["kpp"],
                "contract_amount": t["contract"]["amount"],
                "contract_subject": t["contract"]["subject"],
                "date_start": t["contract"]["date_start"],
                "date_end": t["contract"]["date_end"]
            })

def batch_sync_completed_tenders():
    """
    Массовая выгрузка завершённых тендеров за период
    """
    # Получаем список завершённых тендеров
    response = requests.get(
        f"{FABRIKANT_API}tenders/completed",
        headers={"Authorization": f"Bearer {API_KEY}"},
        params={
            "date_from": "2025-07-01",
            "date_to": datetime.now().strftime("%Y-%m-%d"),
            "org_inn": "0274010000"
        }
    )
    
    tender_ids = [t["id"] for t in response.json().get("tenders", [])]
    
    # Выгружаем детали каждого
    all_tenders = []
    for tid in tender_ids:
        try:
            data = fetch_tender_data(tid)
            all_tenders.append(data)
            print(f"✅ Тендер #{tid} выгружен")
        except Exception as e:
            print(f"❌ Ошибка #{tid}: {e}")
    
    # Экспорт в CSV
    export_to_csv(all_tenders, "tenders_import_1c.csv")
    
    print(f"\n📊 Итого: {len(all_tenders)} тендеров экспортировано")
    print(f"💰 Общая сумма контрактов: {sum(t['contract']['amount'] for t in all_tenders):,.2f} ₽")

if __name__ == "__main__":
    batch_sync_completed_tenders()

import openpyxl
from openpyxl.styles import Font, Alignment, PatternFill
from datetime import datetime

def generate_smeta(contract_data: dict, template_path: str) -> str:
    """
    Автоматическая генерация сметы на основе данных контракта
    """
    # Загружаем шаблон
    wb = openpyxl.load_workbook(template_path)
    ws = wb.active
    
    # Заполняем реквизиты
    ws['B5'] = contract_data['title']
    ws['B6'] = contract_data['winner']['name']
    ws['B7'] = f"ИНН {contract_data['winner']['inn']}, КПП {contract_data['winner']['kpp']}"
    ws['B8'] = contract_data['contract']['subject']
    ws['E12'] = contract_data['contract']['amount']
    ws['E14'] = f"с {contract_data['contract']['date_start']} по {contract_data['contract']['date_end']}"
    
    # Форматирование
    ws['E12'].font = Font(bold=True, size=12)
    ws['E12'].fill = PatternFill(start_color="E8F5E9", end_color="E8F5E9", fill_type="solid")
    
    # Сохраняем
    output_path = f"smeta_{contract_data['tender_id']}_{datetime.now().strftime('%Y%m%d')}.xlsx"
    wb.save(output_path)
    
    return output_path
