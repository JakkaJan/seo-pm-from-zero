import requests
import json

BITRIX_WEBHOOK = "https://company.bitrix24.ru/rest/1/TOKEN/"
ONEC_API = "https://1c.company.ru/api/documents"

def sync_purchase_requests():
    """Заявки из Битрикс24 → 1С"""
    response = requests.get(
        f"{BITRIX_WEBHOOK}crm.item.list",
        params={"entityTypeId": 171, "filter": {"stageId": "NEW"}}
    )
    
    for req in response.json().get("result", {}).get("items", []):
        payload = {
            "document_type": "ТребованиеНаСклад",
            "items": [{
                "name": req["title"],
                "quantity": req["ufCrmQuantity"],
                "unit": req["ufCrmUnit"]
            }]
        }
        requests.post(ONEC_API, json=payload)
        print(f"✅ Заявка #{req['id']} синхронизирована")

sync_purchase_requests()
